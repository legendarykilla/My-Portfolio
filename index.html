YoY Revenue Growth% =
VAR CurrRev = [Total Revenue]
VAR PrevRev = CALCULATE(
    [Total Revenue],
    SAMEPERIODLASTYEAR(
        'Calendar'[Date]
    )
)
RETURN
DIVIDE( CurrRev - PrevRev, PrevRev, 0 )

Rolling 30D Avg =

    DATESINPERIOD(
        'Calendar'[Date],
        MAX('Calendar'[Date]),
        -30, DAY
    ),
    [Daily Revenue]
